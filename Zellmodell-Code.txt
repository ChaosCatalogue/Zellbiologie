      const sharedAnimationMaterials = new Set(Object.values(animationMaterials));
      function clearFunctionalAnimation() {
        animationRoot.traverse((object) => {
          if (!object.isMesh && !object.isSprite) return;
          if (object.geometry && object.geometry !== animationGeometry) object.geometry.dispose();
          if (object.material) {
            const materials = Array.isArray(object.material) ? object.material : [object.material];
            for (const material of materials) {

      const tissueToggle = document.querySelector("#tissue-toggle");
      const tissueStatus = document.querySelector("#tissue-status");
      const savedView = { camera: null, target: null };
      let tissueMode = false;
      let tissuePreviousView = null;
      const selectedWorldPosition = new THREE.Vector3();
      const controlPanel = document.querySelector(".panel");
      let selectedOrganelle = null;
      let restoringView = false;
      let explorationMode = false;
      let freeYaw = 0;
      let freePitch = 0;
      let freeViewDistance = 1;
      let lookPointer = null;
      const movementKeys = new Set();
      let destinationCameraPosition = camera.position.clone();
      let destinationTarget = controls.target.clone();
      let transitioning = false;

      function positionDetailPanel() {
        if (detailPanel.hidden) return;
        if (window.innerWidth <= 1200) {
          detailPanel.style.removeProperty("left");
          detailPanel.style.removeProperty("right");
          detailPanel.style.removeProperty("top");
          return;
        }

        const projected = selectedWorldPosition.clone().project(camera);
        const margin = 16;
        const panelWidth = detailPanel.offsetWidth;
        const panelHeight = detailPanel.offsetHeight;
        const anchorY = ((1 - projected.y) / 2) * window.innerHeight;
        const left = window.innerWidth - panelWidth - margin;
        const top = THREE.MathUtils.clamp(
          anchorY - panelHeight / 2,
          margin,
          Math.max(margin, window.innerHeight - panelHeight - margin)
        );
        detailPanel.style.left = `${left}px`;
        detailPanel.style.right = "auto";
        detailPanel.style.top = `${top}px`;
      }

      function setTissueMode(active) {
        if (tissueMode === active) return;
        tissueMode = active;
        tissueContext.visible = active;
        tissueToggle.setAttribute("aria-pressed", String(active));
        tissueToggle.textContent = active ? "Gewebeansicht beenden" : "Gewebeansicht starten";
        tissueStatus.hidden = !active;
        if (active) {
          tissuePreviousView = {
            camera: destinationCameraPosition.clone(),
            target: destinationTarget.clone()
          };
          controls.maxDistance = 12;
          destinationTarget.set(0, 0, 0);
          destinationCameraPosition.set(0, 0.2, 10.8);
        } else if (tissuePreviousView) {
          controls.maxDistance = 8;
          destinationCameraPosition.copy(tissuePreviousView.camera);
          destinationTarget.copy(tissuePreviousView.target);
          tissuePreviousView = null;
        }
        transitioning = true;
      }

      function focusOrganelle(id, anchor = focusAnchors[id], hitPoint = null) {
        const info = details[id];
        if (!info || !anchor) return;
        if (explorationMode) setExplorationMode(false);

        if (selectedOrganelle === null && !restoringView) {
          savedView.camera = camera.position.clone();
          savedView.target = controls.target.clone();
        }
        restoringView = false;
        selectedOrganelle = id;
        updateProcessOptions(id);
        processLabel.textContent = `Passende Prozesse: ${info.name}`;
        detailPanel.hidden = false;
        controlPanel.classList.toggle("compact", window.innerWidth <= 1200);
        detailPanel.querySelector(".eyebrow").textContent =
          id === "cytoplasm" ? "Zellbereich im Fokus" : "Organell im Fokus";
        document.querySelector("#detail-title").textContent = info.name;
        document.querySelector("#detail-structure").textContent = info.structure;
        document.querySelector("#detail-function").textContent = info.function;
        document.querySelector("#detail-size").textContent = info.size;
        document.querySelector("#detail-volume").textContent = info.volume;
        document.querySelector("#process-title").textContent = "Funktion im 3D-Modell";
        document.querySelector("#process-caption").textContent = info.caption;
        for (const button of organelleList.querySelectorAll("button")) {
          button.setAttribute("aria-pressed", String(button.dataset.organelle === id));
        }

        const target = hitPoint ? hitPoint.clone() : new THREE.Vector3();
        if (!hitPoint) anchor.getWorldPosition(target);
        selectedWorldPosition.copy(target);
        configureFunctionalAnimation(id, target, anchor);
        const direction = camera.position.clone().sub(controls.target).normalize();
        const focusDistances = {
          membrane: 0.9,
          cytoplasm: 0.72,
          nucleus: 0.85,
          mitochondria: 0.48,
          er: 1.05,
          golgi: 0.58,
          ribosomes: 0.38,
          lysosomes: 0.48,
          cytoskeleton: 1.1,
          vesicles: 0.42,
          peroxisomes: 0.48,
          motorProteins: 0.45
        };
        const focusDistance = focusDistances[id];
        const viewTarget = target.clone();
        const cameraRight = new THREE.Vector3(1, 0, 0).applyQuaternion(camera.quaternion).normalize();
        const cameraUp = new THREE.Vector3(0, 1, 0).applyQuaternion(camera.quaternion).normalize();
        const worldPerPixel =
          focusDistance * 2 * Math.tan(THREE.MathUtils.degToRad(camera.fov / 2)) / window.innerHeight;
        if (window.innerWidth <= 1200) {
          const verticalShift = Math.min(window.innerHeight * 0.2, detailPanel.offsetHeight * 0.45);
          viewTarget.addScaledVector(cameraUp, -verticalShift * worldPerPixel);
        } else {
          const margin = 16;
          const panelLeft = window.innerWidth - detailPanel.offsetWidth - margin;
          const navigationRight = controlPanel.getBoundingClientRect().right;
          const desiredScreenX = (navigationRight + panelLeft) / 2;
          const viewportCenterX = window.innerWidth / 2;
          viewTarget.addScaledVector(cameraRight, (viewportCenterX - desiredScreenX) * worldPerPixel);
        }
        destinationTarget = viewTarget;
        destinationCameraPosition = viewTarget.clone().add(direction.multiplyScalar(focusDistance));
        transitioning = true;
      }

      processSelect.addEventListener("change", () => {
        const processId = processSelect.value;
        const process = cellProcesses[processId];
        if (!process) {
          if (tissueMode) setTissueMode(false);
          if (selectedOrganelle !== null) {
            const info = details[selectedOrganelle];
            document.querySelector("#process-title").textContent = "Funktion im 3D-Modell";
            document.querySelector("#process-caption").textContent = info.caption;
            configureFunctionalAnimation(
              selectedOrganelle,
              selectedWorldPosition,
              focusAnchors[selectedOrganelle]
            );
          }
          return;
        }

        if (tissueMode && !process.tissue) setTissueMode(false);
        focusOrganelle(process.focus);
        document.querySelector("#process-title").textContent = process.title;
        document.querySelector("#process-caption").textContent = process.caption;
        configureFunctionalAnimation(
          processId,
          process.globalAnimation ? new THREE.Vector3() : selectedWorldPosition,
          focusAnchors[process.focus],
          process.focus
        );
        if (process.tissue) {
          if (tissueMode) setTissueMode(false);
          setTissueMode(true);
        }
      });

      function setExplorationMode(active) {
        if (explorationMode === active) return;
        if (active && tissueMode) setTissueMode(false);
        if (active && selectedOrganelle !== null) closeDetail();
        explorationMode = active;
        controls.enabled = !active;
        exploreToggle.setAttribute("aria-pressed", String(active));
        exploreToggle.textContent = active ? "Innenansicht beenden" : "Innenansicht starten";
        exploreHelp.hidden = !active;
        movementKeys.clear();
        lookPointer = null;
        if (active) {
          transitioning = false;
          restoringView = false;
          const direction = camera.getWorldDirection(new THREE.Vector3());
          freeYaw = Math.atan2(-direction.x, -direction.z);
          freePitch = Math.asin(THREE.MathUtils.clamp(direction.y, -0.98, 0.98));
          freeViewDistance = Math.max(camera.position.distanceTo(controls.target), 0.5);
          renderer.domElement.focus();
        }
      }

      function closeDetail() {
        if (selectedOrganelle === null) return;
        if (tissueMode) setTissueMode(false);
        selectedOrganelle = null;
        processSelect.value = "";
        updateProcessOptions();
        processLabel.textContent = "Zellprozess im 3D-Modell";
        detailPanel.hidden = true;
        controlPanel.classList.remove("compact");
        animationRoot.visible = false;
        clearFunctionalAnimation();
        focusHalo.visible = false;
        animationUpdater = () => {};
        for (const button of organelleList.querySelectorAll("button")) {
          button.setAttribute("aria-pressed", "false");
        }
        restoringView = true;
        destinationCameraPosition = savedView.camera
          ? savedView.camera.clone()
          : initialCameraPosition.clone();
        destinationTarget = savedView.target
          ? savedView.target.clone()
          : new THREE.Vector3(0, 0, 0);
        transitioning = true;
      }

      let pointerStart = null;
      renderer.domElement.addEventListener("pointerdown", (event) => {
        if (explorationMode) {
          if (event.button === 0) {
            lookPointer = { id: event.pointerId, x: event.clientX, y: event.clientY };
            renderer.domElement.setPointerCapture(event.pointerId);
          }
          return;
        }
        if (event.button !== 0) return;
        pointerStart = { x: event.clientX, y: event.clientY };
      });
      renderer.domElement.addEventListener("pointermove", (event) => {
        if (!explorationMode || !lookPointer || event.pointerId !== lookPointer.id) return;
        const deltaX = event.clientX - lookPointer.x;
        const deltaY = event.clientY - lookPointer.y;
        lookPointer.x = event.clientX;
        lookPointer.y = event.clientY;
        freeYaw -= deltaX * 0.003;
        freePitch = THREE.MathUtils.clamp(freePitch - deltaY * 0.003, -1.5, 1.5);
        const direction = new THREE.Vector3(
          -Math.sin(freeYaw) * Math.cos(freePitch),
          Math.sin(freePitch),
          -Math.cos(freeYaw) * Math.cos(freePitch)
        );
        controls.target.copy(camera.position).addScaledVector(direction, freeViewDistance);
        camera.lookAt(controls.target);
      });
      renderer.domElement.addEventListener("pointerup", (event) => {
        if (explorationMode) {
          if (lookPointer && renderer.domElement.hasPointerCapture(event.pointerId)) {
            renderer.domElement.releasePointerCapture(event.pointerId);
          }
          lookPointer = null;
          return;
        }
        if (!pointerStart) return;
        const moved = Math.hypot(
          event.clientX - pointerStart.x,
          event.clientY - pointerStart.y
        );
        pointerStart = null;
        if (moved > 5) return;

        const bounds = renderer.domElement.getBoundingClientRect();
        const pointer = new THREE.Vector2(
          ((event.clientX - bounds.left) / bounds.width) * 2 - 1,
          -((event.clientY - bounds.top) / bounds.height) * 2 + 1
        );
        const raycaster = new THREE.Raycaster();
        raycaster.setFromCamera(pointer, camera);
        const intersections = raycaster.intersectObjects(interactiveMeshes, false);
        const hit = intersections.find(
          (intersection) => intersection.object.userData.organelleId !== "membrane"
        ) || intersections[0];
        if (hit) {
          if (tissueMode) setTissueMode(false);
          processSelect.value = "";
          focusOrganelle(
            hit.object.userData.organelleId,
            hit.object.userData.organelleAnchor,
            hit.object.userData.organelleId === "membrane" ? hit.point : null
          );
        }
      });
      renderer.domElement.addEventListener("pointerleave", () => { pointerStart = null; });
      renderer.domElement.addEventListener("contextmenu", (event) => event.preventDefault());
      renderer.domElement.addEventListener("wheel", (event) => {
        if (!explorationMode) return;
        event.preventDefault();
        const direction = camera.getWorldDirection(new THREE.Vector3());
        const distance = -Math.sign(event.deltaY) * 0.18;
        camera.position.addScaledVector(direction, distance);
        controls.target.addScaledVector(direction, distance);
      }, { passive: false });

      document.querySelector("#close-detail").addEventListener("click", closeDetail);
      exploreToggle.addEventListener("click", () => setExplorationMode(!explorationMode));
      tissueToggle.addEventListener("click", () => {
        if (tissueMode && cellProcesses[processSelect.value]?.tissue) {
          processSelect.value = "";
          processSelect.dispatchEvent(new Event("change"));
          return;
        }
        setTissueMode(!tissueMode);
      });
      window.addEventListener("keydown", (event) => {
        if (!explorationMode) return;
        const key = event.key.toLowerCase();
        if (["w", "a", "s", "d", "q", "e", "arrowup", "arrowdown", "arrowleft", "arrowright"].includes(key)) {
          event.preventDefault();
          movementKeys.add(key);
        }
      });
      window.addEventListener("keyup", (event) => movementKeys.delete(event.key.toLowerCase()));
      window.addEventListener("blur", () => movementKeys.clear());
      document.querySelector("#reset-view").addEventListener("click", () => {
        setExplorationMode(false);
        if (tissueMode) setTissueMode(false);
        closeDetail();
        if (selectedOrganelle === null) {
          destinationCameraPosition = initialCameraPosition.clone();
          destinationTarget.set(0, 0, 0);
          transitioning = true;
        }
      });

      window.addEventListener("resize", () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        controlPanel.classList.toggle("compact", selectedOrganelle !== null && window.innerWidth <= 1200);
      });

      const clock = new THREE.Clock();
      const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
      const motorTrackAxis = new THREE.Vector3(1, 0, 0);
      let elapsed = 0;
      renderer.setAnimationLoop(() => {
        const delta = Math.min(clock.getDelta(), 0.05);
        elapsed += delta;
        if (!reduceMotion) {
          for (const carrier of movingCarriers) {
            if (selectedOrganelle !== "motorProteins" && selectedOrganelle !== "vesicles") {
              const progress = (elapsed * 0.045 + carrier.offset) % 1;
              carrier.carrier.position.copy(carrier.route.getPointAt(progress));
              const tangent = carrier.route.getTangentAt(progress).normalize();
              carrier.carrier.quaternion.setFromUnitVectors(motorTrackAxis, tangent);
            }
            for (const head of carrier.heads) {
              head.mesh.position.set(
                head.side * Math.sin(elapsed * 8 + carrier.offset) * 0.009,
                head.side * 0.016,
                0
              );
            }
          }
          for (const cisterna of golgiCisternae) {
            const wave = Math.sin(elapsed * 1.15 + cisterna.phase);
            cisterna.group.position.set(
              cisterna.basePosition.x + Math.sin(elapsed * 0.8 + cisterna.phase) * 0.0025,
              cisterna.basePosition.y + wave * 0.002,
              cisterna.basePosition.z + Math.cos(elapsed * 0.7 + cisterna.phase) * 0.0025
            );
            cisterna.group.rotation.z = cisterna.baseRotation + wave * 0.018;
            cisterna.group.scale.set(
              1 + wave * 0.014,
              1 + Math.cos(elapsed + cisterna.phase) * 0.018,
              1 + Math.sin(elapsed * 0.9 + cisterna.phase) * 0.022
            );
          }
          for (const vesicle of secretoryVesicles) {
            const progress = (elapsed * 0.11 + vesicle.phaseOffset) % 1;
            const routeProgress = Math.min(progress / cargoReleaseProgress, 1);
            vesicle.group.position.copy(secretoryRoute.getPointAt(routeProgress));
            const fade = THREE.MathUtils.clamp(
              (progress - cargoReleaseProgress) / 0.08,
              0,
              1
            );
            vesicle.shell.scale.setScalar(1 - fade);
            const releaseDistance = Math.max(0, progress - cargoReleaseProgress);
            const outwardDirection = secretoryRoute.getTangentAt(1).normalize();
            vesicle.cargoParticles.forEach((cargo, index) => {
              const spread = index - 1;
              cargo.position.set(
                spread * 0.012 + outwardDirection.x * releaseDistance * 1.8,
                (spread % 2) * 0.014 + outwardDirection.y * releaseDistance * 1.8,
                0.008 + outwardDirection.z * releaseDistance * 1.8
              );
            });
          }
        }
        if (explorationMode && movementKeys.size > 0) {
          const forward = camera.getWorldDirection(new THREE.Vector3());
          forward.y = 0;
          forward.normalize();
          const right = new THREE.Vector3().crossVectors(forward, camera.up).normalize();
          const movement = new THREE.Vector3();
          if (movementKeys.has("w") || movementKeys.has("arrowup")) movement.add(forward);
          if (movementKeys.has("s") || movementKeys.has("arrowdown")) movement.sub(forward);
          if (movementKeys.has("d") || movementKeys.has("arrowright")) movement.add(right);
          if (movementKeys.has("a") || movementKeys.has("arrowleft")) movement.sub(right);
          if (movementKeys.has("e")) movement.y += 1;
          if (movementKeys.has("q")) movement.y -= 1;
          if (movement.lengthSq() > 0) {
            const speed = (movementKeys.has("shift") ? 2.4 : 0.85) * delta;
            movement.normalize().multiplyScalar(speed);
            camera.position.add(movement);
            controls.target.add(movement);
          }
        }
        if (transitioning) {
          const ease = reduceMotion ? 1 : 1 - Math.exp(-7 * delta);
          camera.position.lerp(destinationCameraPosition, ease);
          controls.target.lerp(destinationTarget, ease);
          if (
            camera.position.distanceToSquared(destinationCameraPosition) < 0.0004 &&
            controls.target.distanceToSquared(destinationTarget) < 0.0004
          ) {
            camera.position.copy(destinationCameraPosition);
            controls.target.copy(destinationTarget);
            transitioning = false;
            restoringView = false;
          }
        }
        if (!reduceMotion && animationRoot.visible) animationUpdater(elapsed);
        if (focusHalo.visible) {
          focusHalo.lookAt(camera.position);
          const pulse = reduceMotion ? 1 : 1 + Math.sin(elapsed * 2) * 0.035;
          focusHalo.scale.setScalar(haloSizes[selectedOrganelle] * pulse);
        }
        if (!explorationMode) controls.update();
        positionDetailPanel();
        renderer.render(scene, camera);
      });
    } catch (error) {
      errorElement.hidden = false;
      errorElement.textContent =
        "Das 3D-Modell konnte nicht gestartet werden. Bitte öffne diese Datei über einen lokalen Webserver und prüfe deine Internetverbindung. " +
        error.message;
      console.error(error);
    }
  </script>
</body>
